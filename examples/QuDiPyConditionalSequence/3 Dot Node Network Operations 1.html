all_pulses = {}

for key in pulse_train[-1].ctrl_pulses.keys():
    all_pulses[f'{key}'] = np.array([])


for pulses in pulse_train:

    if isinstance(pulses, ControlPulse):
        
        for key in pulses.ctrl_pulses.keys():
            # all_pulses[f'{key}'].append(pulses.ctrl_pulses[f'{key}'])

            all_pulses[f'{key}'] = np.hstack((all_pulses[f'{key}'], pulses.ctrl_pulses[f'{key}']))

    else:

        for pulse in pulses:

from qudipy.utils import constants

n_pts = len(all_pulses['V_1'])
t = np.linspace(0, np.max(rot_data['time']), n_pts)*1e6

ctrl_pulses = {'V':{},'W':{},'dg':{},'J':{},'Brf':{} }

for idx, key in enumerate(all_pulses.keys()):
    if 'f_rf' not in key and 'zeeman' not in key and 'phi' not in key: 
        if 'V' in key:
            ctrl_pulses['V'][key] = all_pulses[key]
        elif 'W' in key:
            ctrl_pulses['W'][key] = all_pulses[key]
        elif 'delta' in key:
            ctrl_pulses['dg'][key] = all_pulses[key]
        elif 'J' in key:
            ctrl_pulses['J'][key] = all_pulses[key]
        elif 'rf' in key:
            ctrl_pulses['Brf'][key] = all_pulses[key]


fig, ax = plt.subplots(len(ctrl_pulses.keys()),1, figsize=(15,12))

font = {'size'   : 16}
plt.rc('font', **font)

linthickness = 3

for idx, param in enumerate(ctrl_pulses.keys()):
    for key in ctrl_pulses[param].keys():


        if 'V' in key or 'W' in key or 'J' in key or 'B' in key:

            strings = key.split('_')
            base = strings[0]
            subscript = strings[-1]

            pulse_label = f'${base}_{{{subscript}}}$'

            if 'J' in key:
                ax[idx].plot(t, (1e9/constants.e)*ctrl_pulses[param][key], label = pulse_label, linewidth=linthickness)
                ax[idx].set_ylabel(f'{base} [neV]')
                ax[idx].legend(ncol=6, loc='upper right')
            elif 'B' in key:
                ax[idx].plot(t, 1e6*ctrl_pulses[param][key], linewidth=linthickness)
                ax[idx].set_ylabel('$B_{rf}$ [$\mu$T]')
            else:

                ax[idx].plot(t, ctrl_pulses[param][key], label = pulse_label, linewidth=linthickness)
                ax[idx].set_ylabel(f'{base} [Volts]')

                if 'V' in key:
                    ax[idx].legend(ncol=6, loc='upper left')
                else:
                    ax[idx].legend(ncol=6, loc='upper right')


        elif 'delta' in key:

            strings = key.split('_')
            base = strings[0]
            subscript = strings[-1]

            pulse_label = f'$\{strings[0]} {strings[1]}_{{{strings[-1]}}}$'

            ax[idx].plot(t, ctrl_pulses[param][key], label = pulse_label, linewidth=linthickness)
            ax[idx].set_ylabel('$\delta g$')
